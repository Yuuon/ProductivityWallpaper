Recommended implementation:
```csharp
private async Task<string> GenerateVideoThumbnailAsync(string sourcePath, string outputPath, CancellationToken ct)
{
    // Use LibVLC if available (from WallpaperService pattern)
    // Or use System.Drawing with Windows API
    
    // Simplest reliable approach for WPF:
    // Use MediaPlayer to render frame to RenderTargetBitmap
    var player = new MediaPlayer();
    player.Open(new Uri(sourcePath));
    
    // Wait for media opened
    await Task.Delay(500, ct);
    
    // Create drawing visual
    var visual = new DrawingVisual();
    using (var context = visual.RenderOpen())
    {
        context.DrawVideo(player, new Rect(0, 0, ThumbnailWidth, ThumbnailHeight));
    }
    
    var bitmap = new RenderTargetBitmap(ThumbnailWidth, ThumbnailHeight, 96, 96, PixelFormats.Pbgra32);
    bitmap.Render(visual);
    
    // Encode and save as JPEG
    var encoder = new JpegBitmapEncoder { QualityLevel = ThumbnailQuality };
    encoder.Frames.Add(BitmapFrame.Create(bitmap));
    
    using var stream = File.Create(outputPath);
    encoder.Save(stream);
    
    player.Close();
    return outputPath;
}
```
